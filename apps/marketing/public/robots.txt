# AEGIS — robots policy for aegistraces.com
#
# We want to be cited by AI answer engines (ChatGPT, Claude, Perplexity,
# Gemini, Bing Copilot). The three surfaces they crawl are:
#   1. sitemap.xml   — page index with lastmod + priority
#   2. llms.txt      — human-authored map for LLM crawlers per the
#                      emerging llms.txt convention
#   3. llms-full.txt — single-file concatenation of every blog article
#                      (preferred over follow-the-link crawling)

# ── Default policy ─────────────────────────────────────────────────
User-agent: *
Allow: /
Disallow: /api/
Disallow: /account
Disallow: /login
Disallow: /signup
Crawl-delay: 1

# ── AI answer-engine crawlers (explicit allow) ─────────────────────
# These bots read structured content to answer user questions in
# real time. We want them citing us. Explicit User-agent entries
# take precedence over the wildcard above for these agents.
